export type Primitive =
    string |
    number |
    boolean |
    bigint |
    symbol |
    null |
    undefined;

export interface MapfunCompatible<T = any> {
    __mapfun__?: boolean;
}

export interface ComplexLike {
    isComplex(): boolean;
}

export interface MatrixLike {
    rows: number;
    cols: number;

    isMatrix(): boolean;
}

export type Mappable<T = any> =
    | T
    | Primitive
    | Mappable<T>[]
    | ArrayLike<T>
    | Set<Mappable<T>>
    | Map<any, Mappable<T>>
    | MapfunCompatible<T>
    | ComplexLike
    | MatrixLike
    | Record<string, Mappable<T>>;


/**
 * Applies a function recursively to values inside
 * arrays, typed arrays, sets, maps, objects,
 * complex values and matrices.
 *
 * @param fun Transformation function.
 * @param X Values to transform.
 *
 * @example
 * mapfun(x => x * 2, [1,2,3])
 * // [2,4,6]
 */
export declare function mapfun<T, R>(
    fun: (value: T) => R,
    x: Mappable<T>
): R;

export declare function mapfun<T, R>(
    fun: (value: T) => R,
    x: Mappable<T>,
    ...X: Mappable<T>[]
): R[];


/**
 * Applies a function to supported mathematical structures.
 *
 * Handles Complex and Matrix instances specially.
 *
 * @param x Input value.
 * @param fn Transformation function.
 */
export declare function apply_fun<T, R>(
    x: Mappable<T>,
    fn: (value: T) => R
): Mappable<R>;
